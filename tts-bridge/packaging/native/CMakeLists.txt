cmake_minimum_required(VERSION 3.20)
project(bridge_supervisor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(bridge-supervisor supervisor.cpp)
if(WIN32)
  target_link_libraries(bridge-supervisor PRIVATE iphlpapi ws2_32)
endif()
if(MSVC)
  target_compile_options(bridge-supervisor PRIVATE /utf-8)
  set_property(TARGET bridge-supervisor PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
else()
  target_compile_options(bridge-supervisor PRIVATE "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=native" "-fdebug-prefix-map=${CMAKE_BINARY_DIR}=build")
endif()
